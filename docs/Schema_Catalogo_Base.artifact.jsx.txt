// ================================================================
//  SCHEMA CATALOGO - BASE DI LAVORO
//  Gerarchia: 1 Brand -> 2 Tipo Cliente -> 3 Categoria -> 4 Prodotto -> 5 Offerta -> 6 Opzioni
//
//  Struttura del blocco CATALOGO:
//     CATALOGO[brandId][tipoCliente][categoria] = [
//       { nome: "Prodotto", offerte: [
//           "Offerta semplice",
//           { nome: "Offerta con opzioni", opzioni: ["Opzione A", "Opzione B"] }
//       ] },
//       ...
//     ]
//  Se un brand non ha un tipo cliente o una categoria, semplicemente
//  non si scrive la chiave (es. Dojo ha solo Business > POS).
//  Prodotti e offerte scritti come semplici stringhe sono comunque
//  accettati (trattati come privi di offerte / opzioni).
//  Le opzioni (Livello 6) sono a SELEZIONE MULTIPLA sulla singola offerta,
//  a differenza dei livelli superiori che sono a selezione singola.
//  Un'opzione puo' anche essere { nome, tipo: "numero" }: quando viene
//  selezionata chiede una quantita' (es. Stanze Aggiuntive su Sky TV Hotel).
// ================================================================

// ---------------------------------------------------------------
//  Helpers (nessun optional chaining, controlli espliciti)
// ---------------------------------------------------------------
function getRamo(brandId, tipo) {
  const b = CATALOGO[brandId];
  if (!b) return null;
  const t = b[tipo];
  return t ? t : null;
}

function tipoDisponibile(brandId, tipo) {
  return getRamo(brandId, tipo) !== null;
}

function brandHaCategoria(brandId, tipo, categoria) {
  const ramo = getRamo(brandId, tipo);
  if (!ramo) return false;
  const arr = ramo[categoria];
  return arr && arr.length > 0 ? true : false;
}

function getProdotti(brandId, tipo, categoria) {
  const ramo = getRamo(brandId, tipo);
  if (!ramo) return [];
  const out = [];
  CATEGORIE.forEach(function (c) {
    if (categoria && c !== categoria) return;
    const arr = ramo[c];
    if (arr) {
      arr.forEach(function (p) {
        if (typeof p === "string") {
          out.push({ categoria: c, nome: p, offerte: [] });
        } else {
          const offGrezze = p.offerte ? p.offerte : [];
          const offNorm = offGrezze.map(function (o) {
            if (typeof o === "string") return { nome: o, opzioni: [] };
            const opzGrezze = o.opzioni ? o.opzioni : [];
            const opzNorm = opzGrezze.map(function (op) {
              if (typeof op === "string") return { nome: op, tipo: null };
              return { nome: op.nome, tipo: op.tipo ? op.tipo : null };
            });
            return { nome: o.nome, opzioni: opzNorm };
          });
          out.push({ categoria: c, nome: p.nome, offerte: offNorm });
        }
      });
    }
  });
  return out;
}

function contaProdotti(brandId, tipo, categoria) {
  return getProdotti(brandId, tipo, categoria).length;
}

function contaOfferte(brandId, tipo, categoria) {
  const ps = getProdotti(brandId, tipo, categoria);
  let n = 0;
  ps.forEach(function (p) { n = n + p.offerte.length; });
  return n;
}

function chiaveProdotto(p) {
  return p.categoria + "|" + p.nome;
}

function risolviCampi(brandId, tipoCliente, categoria, prodottoNome, offertaNome, opzNomi) {
  const out = [];
  const visti = {};
  CAMPI_REGOLE.forEach(function (r) {
    if (r.brand && r.brand.indexOf(brandId) === -1) return;
    if (r.tipo && r.tipo.indexOf(tipoCliente) === -1) return;
    if (r.categoria && r.categoria.indexOf(categoria) === -1) return;
    if (r.prodotto && r.prodotto.indexOf(prodottoNome) === -1) return;
    if (r.offertaNon && r.offertaNon.indexOf(offertaNome) !== -1) return;
    if (r.offertaContiene) {
      const low = offertaNome.toLowerCase();
      const trovato = r.offertaContiene.some(function (s) { return low.indexOf(s.toLowerCase()) !== -1; });
      if (!trovato) return;
    }
    if (r.opzioni) {
      const attiva = r.opzioni.some(function (o) { return opzNomi.indexOf(o) !== -1; });
      if (!attiva) return;
    }
    r.campi.forEach(function (cmp) {
      if (visti[cmp.nome]) return;
      visti[cmp.nome] = true;
      out.push(cmp);
    });
  });
  return out;
}

              fontWeight: 600,
              cursor: disp ? "pointer" : "not-allowed",
              fontFamily: "inherit"
            }}
          >
            {t}
          </button>
        );
        return btn;
      })}
    </div>
  );

  // ------- Livello 3: categorie (chip filtro) -------
  const chips = [null].concat(CATEGORIE);
  const chipsUI = (
    <div style={{ display: "flex", gap: 6, flexWrap: "wrap" }}>
      {chips.map(function (c) {
        const isAll = c === null;
        const sel = categoria === c;
        const avail = isAll
          ? contaProdotti(brandId, tipo, null) > 0
          : brandHaCategoria(brandId, tipo, c);
        const n = contaProdotti(brandId, tipo, c);
        const chip = (
          <button
            key={isAll ? "tutte" : c}
            onClick={function () { selCategoria(c); }}
            style={{
              display: "flex",
              alignItems: "center",
              gap: 6,
              padding: "7px 12px",
              borderRadius: 999,
              border: sel ? "1px solid " + brand.c1 : bordo,
              background: sel ? brand.c1 + "22" : "rgba(255,255,255,0.02)",
              color: sel ? "#ffffff" : avail ? "#c6cbd8" : "rgba(255,255,255,0.22)",
              fontSize: 12,
              fontWeight: 600,
              cursor: "pointer",
              fontFamily: "inherit"
            }}
          >
            {isAll ? "Tutte" : c}
            {avail ? (
              <span style={{
                fontSize: 10, fontWeight: 700, padding: "1px 6px", borderRadius: 99,
                background: "rgba(255,255,255,0.08)",
                color: sel ? "#ffffff" : txtMuted
              }}>{n}</span>
            ) : null}
          </button>
        );
        return chip;
      })}
    </div>
  );

  // ------- Livello 4: prodotti (selezionabili) -------
  const gruppi = [];
  CATEGORIE.forEach(function (cat) {
    const items = prodotti.filter(function (p) { return p.categoria === cat; });
    if (items.length > 0) gruppi.push({ cat: cat, items: items });
  });

  const briciole = brand.nome + " \u00b7 " + tipo + (categoria ? " \u00b7 " + categoria : "");

  let listaUI = null;
  if (prodotti.length === 0) {
    listaUI = (
      <div style={{ padding: "30px 16px", textAlign: "center", color: txtMuted, fontSize: 13 }}>
        Nessun prodotto per <b style={{ color: "#e8eaf0" }}>{briciole}</b>: combinazione non prevista o ancora da compilare.
      </div>
    );
  } else {
    listaUI = (
      <div style={{ display: "flex", flexDirection: "column", gap: 14 }}>
        {gruppi.map(function (g) {
          const blocco = (
            <div key={g.cat}>
              {categoria === null ? (
                <div style={{
                  fontSize: 10, fontWeight: 700, color: txtMuted,
                  textTransform: "uppercase", letterSpacing: 1, margin: "2px 0 6px"
                }}>{g.cat}</div>
              ) : null}
              <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fill, minmax(200px, 1fr))", gap: 6 }}>
                {g.items.map(function (p) {
                  const k = chiaveProdotto(p);
                  const selP = prodSel === k;
                  const nOff = p.offerte.length;
                  const riga = (
                    <div
                      key={k}
                      onClick={function () { selProdotto(k); }}
                      style={{
                        display: "flex", alignItems: "center", gap: 10,
                        padding: "10px 12px", borderRadius: 10,
                        cursor: "pointer",
                        background: selP ? brand.c1 + "1c" : "rgba(255,255,255,0.03)",
                        border: selP ? "1px solid " + brand.c1 : "1px solid rgba(255,255,255,0.06)",
                        transition: "all .15s ease"
                      }}
                    >
                      <span style={{
                        width: 3, height: 16, borderRadius: 2, flexShrink: 0,
                        background: "linear-gradient(180deg, " + brand.c1 + ", " + brand.c2 + ")"
                      }} />
                      <span style={{ fontSize: 13, flex: 1 }}>{p.nome}</span>
                      <span style={{
                        fontSize: 10, fontWeight: 700, padding: "1px 7px", borderRadius: 99,
                        background: selP ? brand.c1 + "33" : "rgba(255,255,255,0.07)",
                        color: nOff === 0 ? "#f5c400" : selP ? "#ffffff" : txtMuted
                      }}>{nOff}</span>
                    </div>
                  );
                  return riga;
                })}
              </div>
            </div>
          );
          return blocco;
        })}
      </div>
    );
  }

  const prodottiUI = (
    <div style={{ background: "#141826", border: bordo, borderRadius: 14, padding: 16 }}>
      <div style={{ display: "flex", alignItems: "baseline", justifyContent: "space-between", marginBottom: 12, flexWrap: "wrap", gap: 8 }}>
        <div style={{ ...etich, marginBottom: 0 }}>Livello 4 &middot; Prodotti</div>
        <div style={{ fontSize: 12, color: txtMuted }}>{briciole} &middot; {prodotti.length} {prodotti.length === 1 ? "prodotto" : "prodotti"}</div>
      </div>
      {listaUI}
      {prodSel === null && prodotti.length > 0 ? (
        <div style={{ marginTop: 14, paddingTop: 12, borderTop: "1px solid rgba(255,255,255,0.05)", fontSize: 11, color: "#5c657c" }}>
          Clic su un prodotto per aprire il Livello 5 &middot; Offerte. Il numero sul badge indica quante offerte contiene.
        </div>
      ) : null}
    </div>
  );

  // ------- Livello 5: offerte del prodotto selezionato -------
  let offerteUI = null;
  if (prodottoSel !== null) {
    const briciole5 = briciole + " \u00b7 " + prodottoSel.nome;
    let corpoOff = null;
    if (prodottoSel.offerte.length === 0) {
      corpoOff = (
        <div style={{ padding: "26px 16px", textAlign: "center", color: txtMuted, fontSize: 13 }}>
          Nessuna offerta per <b style={{ color: "#e8eaf0" }}>{prodottoSel.nome}</b>: da compilare.
        </div>
      );
    } else {
      corpoOff = (
        <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fill, minmax(180px, 1fr))", gap: 6 }}>
          {prodottoSel.offerte.map(function (o, i) {
            const selO = offSel === i;
            const nOpz = o.opzioni.length;
            const rigaOff = (
              <div
                key={i}
                onClick={function () { selOfferta(i); }}
                style={{
                  display: "flex", alignItems: "center", gap: 10,
                  padding: "9px 12px", borderRadius: 10,
                  cursor: "pointer",
                  background: selO ? brand.c1 + "1c" : "rgba(255,255,255,0.03)",
                  border: selO ? "1px solid " + brand.c1 : "1px solid rgba(255,255,255,0.06)",
                  transition: "all .15s ease"
                }}
              >
                <span style={{
                  width: 7, height: 7, borderRadius: 99, flexShrink: 0,
                  border: "2px solid " + brand.c1
                }} />
                <span style={{ fontSize: 13, flex: 1 }}>{o.nome}</span>
                <span style={{
                  fontSize: 10, fontWeight: 700, padding: "1px 7px", borderRadius: 99,
                  background: selO ? brand.c1 + "33" : "rgba(255,255,255,0.07)",
                  color: nOpz === 0 ? "#f5c400" : selO ? "#ffffff" : txtMuted
                }}>{nOpz}</span>
              </div>
            );
            return rigaOff;
          })}
        </div>
      );
    }
    offerteUI = (
      <div style={{ background: "#141826", border: "1px solid " + brand.c1 + "55", borderRadius: 14, padding: 16 }}>
        <div style={{ display: "flex", alignItems: "baseline", justifyContent: "space-between", marginBottom: 12, flexWrap: "wrap", gap: 8 }}>
          <div style={{ ...etich, marginBottom: 0 }}>Livello 5 &middot; Offerte</div>
          <div style={{ fontSize: 12, color: txtMuted }}>{briciole5} &middot; {prodottoSel.offerte.length} {prodottoSel.offerte.length === 1 ? "offerta" : "offerte"}</div>
        </div>
        {corpoOff}
        {offSel === null && prodottoSel.offerte.length > 0 ? (
          <div style={{ marginTop: 12, paddingTop: 10, borderTop: "1px solid rgba(255,255,255,0.05)", fontSize: 11, color: "#5c657c" }}>
            Clic su un'offerta per aprire il Livello 6 &middot; Opzioni. Il badge indica quante opzioni contiene.
          </div>
        ) : null}
      </div>
    );
  }

  // ------- Livello 6: opzioni dell'offerta selezionata (multiselezione) -------
  let opzioniUI = null;
  if (offertaSel !== null) {
    const briciole6 = briciole + " \u00b7 " + prodottoSel.nome + " \u00b7 " + offertaSel.nome;
    let corpoOpz = null;
    if (offertaSel.opzioni.length === 0) {
      corpoOpz = (
        <div style={{ padding: "26px 16px", textAlign: "center", color: txtMuted, fontSize: 13 }}>
          Nessuna opzione per <b style={{ color: "#e8eaf0" }}>{offertaSel.nome}</b>: da compilare.
        </div>
      );
    } else {
      corpoOpz = (
        <div style={{ display: "flex", gap: 6, flexWrap: "wrap" }}>
          {offertaSel.opzioni.map(function (op, i) {
            const selOp = opzSel.indexOf(i) !== -1;
            const chipOp = (
              <span key={i} style={{ display: "inline-flex", alignItems: "center", gap: 6 }}>
                <button
                  onClick={function () { toggleOpzione(i); }}
                  style={{
                    display: "flex", alignItems: "center", gap: 8,
                    padding: "8px 14px", borderRadius: 10,
                    border: selOp ? "1px solid " + brand.c1 : bordo,
                    background: selOp ? brand.c1 + "22" : "rgba(255,255,255,0.02)",
                    color: selOp ? "#ffffff" : "#c6cbd8",
                    fontSize: 12, fontWeight: 600, cursor: "pointer", fontFamily: "inherit"
                  }}
                >
                  <span style={{
                    width: 14, height: 14, borderRadius: 4, flexShrink: 0,
                    border: selOp ? "1px solid " + brand.c1 : "1px solid rgba(255,255,255,0.25)",
                    background: selOp ? brand.c1 : "transparent",
                    display: "inline-flex", alignItems: "center", justifyContent: "center",
                    fontSize: 10, color: "#ffffff", fontWeight: 800
                  }}>{selOp ? "\u2713" : ""}</span>
                  {op.nome}
                  {op.tipo === "numero" ? (
                    <span style={{
                      fontSize: 9, fontWeight: 800, padding: "1px 5px", borderRadius: 99,
                      background: "rgba(255,255,255,0.1)", color: selOp ? "#ffffff" : txtMuted
                    }}>n\u00b0</span>
                  ) : null}
                </button>
                {selOp && op.tipo === "numero" ? (
                  <input
                    type="number"
                    min="1"
                    placeholder="quante?"
                    style={{
                      width: 78, background: "rgba(255,255,255,0.06)",
                      border: "1px solid " + brand.c1, borderRadius: 8,
                      color: "#ffffff", fontSize: 12, padding: "6px 8px", fontFamily: "inherit"
                    }}
                  />
                ) : null}
              </span>
            );
            return chipOp;
          })}
        </div>
      );
    }
    opzioniUI = (
      <div style={{ background: "#141826", border: "1px solid " + brand.c1 + "55", borderRadius: 14, padding: 16 }}>
        <div style={{ display: "flex", alignItems: "baseline", justifyContent: "space-between", marginBottom: 12, flexWrap: "wrap", gap: 8 }}>
          <div style={{ ...etich, marginBottom: 0 }}>Livello 6 &middot; Opzioni</div>
          <div style={{ fontSize: 12, color: txtMuted }}>{briciole6} &middot; {opzSel.length} su {offertaSel.opzioni.length} selezionate</div>
        </div>
        {corpoOpz}
        <div style={{ marginTop: 12, paddingTop: 10, borderTop: "1px solid rgba(255,255,255,0.05)", fontSize: 11, color: "#5c657c" }}>
          A differenza dei livelli precedenti, qui la selezione è multipla: più opzioni possono essere attive sulla stessa offerta.
        </div>
      </div>
    );
  }

  // ------- Strato dati: campi da compilare per la selezione corrente -------
  let campiUI = null;
  if (offertaSel !== null) {
    const opzNomi = opzSel.map(function (i) { return offertaSel.opzioni[i].nome; });
    const campi = risolviCampi(brandId, tipo, prodottoSel.categoria, prodottoSel.nome, offertaSel.nome, opzNomi);
    const coloriTipo = { testo: "#8b93a7", numero: "#2dd4bf", data: "#f5c400", scelta: "#a78bfa" };
    const righeCampi = campi.map(function (cmp, i) {
      const riga = (
        <div key={i} style={{ display: "flex", alignItems: "center", gap: 10, padding: "9px 12px", background: "rgba(255,255,255,0.03)", border: "1px solid rgba(255,255,255,0.07)", borderRadius: 10, flexWrap: "wrap" }}>
          <span style={{ fontSize: 9, fontWeight: 800, letterSpacing: 0.5, textTransform: "uppercase", padding: "3px 8px", borderRadius: 99, background: "rgba(255,255,255,0.06)", color: coloriTipo[cmp.tipo] ? coloriTipo[cmp.tipo] : "#8b93a7", flexShrink: 0, width: 52, textAlign: "center" }}>{cmp.tipo}</span>
          <span style={{ fontSize: 13, fontWeight: 600 }}>{cmp.nome}</span>
          {cmp.nota ? <span style={{ fontSize: 11, color: txtMuted }}>{cmp.nota}</span> : null}
          <span style={{ flex: 1 }} />
          {cmp.conferma ? (
            <span style={{ fontSize: 9, fontWeight: 800, padding: "2px 7px", borderRadius: 99, background: "rgba(255,159,28,0.15)", color: "#ffbe66" }}>da confermare</span>
          ) : null}
        </div>
      );
      return riga;
    });
    campiUI = (
      <div style={{ background: "#10141f", border: "1px dashed rgba(255,255,255,0.2)", borderRadius: 14, padding: 16 }}>
        <div style={{ display: "flex", alignItems: "baseline", justifyContent: "space-between", marginBottom: 12, flexWrap: "wrap", gap: 8 }}>
          <div style={{ ...etich, marginBottom: 0 }}>Dati da Compilare</div>
          <div style={{ fontSize: 12, color: txtMuted }}>{campi.length} campi per questa combinazione di offerta e opzioni</div>
        </div>
        <div style={{ display: "flex", flexDirection: "column", gap: 6 }}>{righeCampi}</div>
        <div style={{ marginTop: 12, paddingTop: 10, borderTop: "1px solid rgba(255,255,255,0.05)", fontSize: 11, color: "#5c657c" }}>
          Non è un settimo livello di selezione: sono i campi che il Registra Vendita chiederà per la combinazione scelta. Attiva o disattiva le opzioni qui sopra per vederli cambiare (es. GNP, RID, Reload Open).
        </div>
      </div>
    );
  }

  // ------- Vista Esplora -------
  const esploraUI = (
    <div style={{ display: "flex", flexDirection: "column", gap: 18 }}>
      {livelliUI}
      <div>
        <div style={etich}>Livello 1 &middot; Brand</div>
        {brandGridUI}
      </div>
      <div style={{ display: "flex", gap: 28, flexWrap: "wrap", alignItems: "flex-start" }}>
        <div>
          <div style={etich}>Livello 2 &middot; Tipo Cliente</div>
          {tipoUI}
        </div>
        <div style={{ flex: 1, minWidth: 320 }}>
          <div style={etich}>Livello 3 &middot; Categoria (anche filtro trasversale)</div>
          {chipsUI}
        </div>
      </div>
      {prodottiUI}
      {offerteUI}
      {opzioniUI}
      {campiUI}
    </div>
  );

  // ------- Vista Matrice -------
  const filtroTipi = ["Tutti"].concat(TIPI);
  const thStyle = {
    padding: "10px 8px",
    fontSize: 10,
    fontWeight: 700,
    color: txtMuted,
    textTransform: "uppercase",
    letterSpacing: 0.5,
    textAlign: "center",
    borderBottom: bordo,
    whiteSpace: "nowrap"
  };

  const matriceUI = (
    <div>
      <div style={{ display: "flex", alignItems: "center", gap: 12, marginBottom: 12, flexWrap: "wrap" }}>
        <div style={{ display: "flex", gap: 6 }}>
          {filtroTipi.map(function (t) {
            const sel = tipoMatrice === t;
            const btn = (
              <button
                key={t}
                onClick={function () { setTipoMatrice(t); }}
                style={{
                  padding: "7px 16px",
                  borderRadius: 10,
                  border: sel ? "1px solid rgba(139,92,246,0.6)" : bordo,
                  background: sel ? "rgba(139,92,246,0.15)" : "transparent",
                  color: sel ? "#c4b5fd" : txtMuted,
                  fontSize: 12,
                  fontWeight: 600,
                  cursor: "pointer",
                  fontFamily: "inherit"
                }}
              >
                {t}
              </button>
            );
            return btn;
          })}
        </div>
        <div style={{ fontSize: 11, color: txtMuted, display: "flex", gap: 8, alignItems: "center" }}>
          <span style={miniC}>C</span> Consumer
          <span style={miniB}>B</span> Business
          <span style={{ marginLeft: 8 }}>Conteggio prodotti &middot; clic su una cella per aprirla in Esplora</span>
        </div>
      </div>

      <div style={{ background: "#141826", border: bordo, borderRadius: 14, overflowX: "auto" }}>
        <table style={{ width: "100%", borderCollapse: "collapse", minWidth: 1080 }}>
          <thead>
            <tr>
              <th style={{ ...thStyle, textAlign: "left", paddingLeft: 16 }}>Brand</th>
              {CATEGORIE.map(function (c) {
                const h = <th key={c} style={thStyle}>{c}</th>;
                return h;
              })}
            </tr>
          </thead>
          <tbody>
            {BRANDS.map(function (b) {
              const row = (
                <tr key={b.id}>
                  <td style={{ padding: "9px 16px", borderBottom: "1px solid rgba(255,255,255,0.05)", whiteSpace: "nowrap" }}>
                    <span style={{
                      display: "inline-block", width: 8, height: 8, borderRadius: 99, marginRight: 8,
                      background: "linear-gradient(135deg, " + b.c1 + ", " + b.c2 + ")"
                    }} />
                    <span style={{ fontSize: 13, fontWeight: 600 }}>{b.nome}</span>
                  </td>
                  {CATEGORIE.map(function (c) {
                    const nC = contaProdotti(b.id, "Consumer", c);
                    const nB = contaProdotti(b.id, "Business", c);
                    let attivo = false;
                    let contenuto = null;
                    if (tipoMatrice === "Tutti") {
                      attivo = nC + nB > 0;
                      contenuto = attivo ? (
                        <span style={{ display: "inline-flex", gap: 4 }}>
                          {nC > 0 ? <span style={miniC}>C {nC}</span> : null}
                          {nB > 0 ? <span style={miniB}>B {nB}</span> : null}
                        </span>
                      ) : (
                        <span style={{ color: "rgba(255,255,255,0.12)" }}>&mdash;</span>
                      );
                    } else {
                      const n = tipoMatrice === "Consumer" ? nC : nB;
                      attivo = n > 0;
                      contenuto = attivo ? (
                        <span style={miniC}>{n}</span>
                      ) : (
                        <span style={{ color: "rgba(255,255,255,0.12)" }}>&mdash;</span>
                      );
                    }
                    const cella = (
                      <td
                        key={c}
                        onClick={attivo ? function () { apriCella(b.id, c); } : undefined}
                        style={{
                          padding: "8px 6px",
                          textAlign: "center",
                          borderBottom: "1px solid rgba(255,255,255,0.05)",
                          cursor: attivo ? "pointer" : "default"
                        }}
                      >
                        {contenuto}
                      </td>
                    );
                    return cella;
                  })}
                </tr>
              );
              return row;
            })}
            <tr>
              <td style={{
                padding: "10px 16px", fontSize: 10, fontWeight: 700,
                color: txtMuted, textTransform: "uppercase", letterSpacing: 0.5, whiteSpace: "nowrap"
              }}>Brand attivi</td>
              {CATEGORIE.map(function (c) {
                let n = 0;
                BRANDS.forEach(function (b) {
                  if (tipoMatrice === "Tutti") {
                    if (contaProdotti(b.id, "Consumer", c) + contaProdotti(b.id, "Business", c) > 0) n = n + 1;
                  } else if (contaProdotti(b.id, tipoMatrice, c) > 0) {
                    n = n + 1;
                  }
                });
                const tot = (
                  <td key={c} style={{
                    padding: "10px 6px", textAlign: "center", fontSize: 12, fontWeight: 700,
                    color: n > 0 ? "#c4b5fd" : "rgba(255,255,255,0.12)"
                  }}>
                    {n > 0 ? n : "\u2014"}
                  </td>
                );
                return tot;
              })}
            </tr>
          </tbody>
        </table>
      </div>
    </div>
  );

  // ------- Assemblaggio -------
  let corpoUI = null;
  if (vista === "esplora") {
    corpoUI = esploraUI;
  } else {
    corpoUI = matriceUI;
  }

  const tabs = [
    { id: "esplora", label: "Esplora" },
    { id: "matrice", label: "Matrice" }
  ];

  const tabsUI = (
    <div style={{ display: "flex", gap: 6 }}>
      {tabs.map(function (t) {
        const sel = vista === t.id;
        const el = (
          <button
            key={t.id}
            onClick={function () { setVista(t.id); }}
            style={{
              padding: "8px 18px",
              borderRadius: 10,
              border: sel ? "1px solid rgba(139,92,246,0.6)" : bordo,
              background: sel ? "rgba(139,92,246,0.15)" : "transparent",
              color: sel ? "#c4b5fd" : txtMuted,
              fontSize: 12,
              fontWeight: 600,
              cursor: "pointer",
              fontFamily: "inherit"
            }}
          >
            {t.label}
          </button>
        );
        return el;
      })}
    </div>
  );

  const ui = (
    <div style={{
      minHeight: "100vh",
      background: "#0b0e14",
      color: "#e8eaf0",
      fontFamily: "'Segoe UI', system-ui, -apple-system, sans-serif",
      padding: "24px 28px",
      boxSizing: "border-box"
    }}>
      <div style={{ maxWidth: 1180, margin: "0 auto" }}>
        <div style={{ display: "flex", alignItems: "flex-start", justifyContent: "space-between", marginBottom: 18, flexWrap: "wrap", gap: 12 }}>
          <div>
            <div style={{ fontSize: 20, fontWeight: 700, letterSpacing: 0.2 }}>Schema Catalogo</div>
            <div style={{ fontSize: 12, color: txtMuted, marginTop: 4 }}>
              Brand &rarr; Tipo Cliente &rarr; Categoria &rarr; Prodotto &rarr; Offerta &rarr; Opzioni
              <span style={{
                marginLeft: 10, fontSize: 10, fontWeight: 700, padding: "2px 8px", borderRadius: 99,
                background: "rgba(245,196,0,0.12)", color: "#f5c400", letterSpacing: 0.5
              }}>OFFERTE DA COMPILARE</span>
            </div>
          </div>
          {tabsUI}
        </div>
        {corpoUI}
      </div>
    </div>
  );
  return ui;
}
